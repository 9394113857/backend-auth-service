# =========================================================
# AUTH SERVICE DATABASE RESET FOR TESTING
# Clears all authentication related tables
# =========================================================


# =========================================================
# STEP 1 — Open Flask shell
# =========================================================
flask shell


# =========================================================
# STEP 2 — Import database and models
# =========================================================

from app.extensions import db
from app.models import (
    User,
    PasswordHistory,
    PasswordResetToken,
    EmailVerificationToken,
    TokenBlocklist
)


# =========================================================
# STEP 3 — Delete all records from tables
# =========================================================

User.query.delete()

PasswordHistory.query.delete()

PasswordResetToken.query.delete()

EmailVerificationToken.query.delete()

TokenBlocklist.query.delete()


# =========================================================
# STEP 4 — Commit changes
# =========================================================

db.session.commit()

print("All auth tables cleared successfully")


# =========================================================
# STEP 5 — Exit Flask shell
# =========================================================

exit()


# =========================================================
# STEP 6 — Restart backend server
# =========================================================

python run.py
