$files = @(
    "run.py",
    "Dockerfile",
    "requirements.txt"
) + (Get-ChildItem -Path app -Recurse -File | Select-Object -ExpandProperty FullName)

"===== BACKEND AUTH SERVICE : FILE CONTENT EXPORT =====" | Out-File -Encoding utf8 APP_CODE_EXPORT.txt

foreach ($file in $files) {
    if (Test-Path $file) {
        "`n==================================================" | Out-File APP_CODE_EXPORT.txt -Append
        "FILE: $file" | Out-File APP_CODE_EXPORT.txt -Append
        "==================================================" | Out-File APP_CODE_EXPORT.txt -Append
        Get-Content $file | Out-File APP_CODE_EXPORT.txt -Append
    }
}
