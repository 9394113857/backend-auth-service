REM ==================================================
REM 🟦 CUSTOMER LOGIN AGAIN + VIEW ORDERS
REM ==================================================
REM Token expired again 😄
REM Generate fresh JWT first
REM ==================================================


REM --------------------------------------------------
REM STEP 1: CUSTOMER LOGIN AGAIN
REM --------------------------------------------------

curl -X POST %AUTH_BASE_URL%/api/v1/auth/angularUser/login ^
-H "Content-Type: application/json" ^
-d "{\"email\":\"%CUSTOMER_EMAIL%\",\"password\":\"%CUSTOMER_PASSWORD%\"}"


REM --------------------------------------------------
REM STEP 2:
REM COPY ONLY access_token VALUE
REM --------------------------------------------------


REM --------------------------------------------------
REM STEP 3: SET NEW CUSTOMER TOKEN
REM --------------------------------------------------

set CUSTOMER_TOKEN=PASTE_NEW_JWT_TOKEN_HERE


REM ==================================================
REM 🟦 FULL FINAL CUSTOMER ORDER FLOW
REM ==================================================
REM Goal:
REM Current Stock = 5
REM Buy Quantity = 2
REM Expected Remaining Stock = 3
REM ==================================================


REM --------------------------------------------------
REM STEP 1: SET PRODUCT DETAILS
REM --------------------------------------------------

set PRODUCT_ID=2
set VARIANT_ID=1
set PRODUCT_NAME=Gaming Laptop Pro
set COLOR=Silver
set PRICE=75000
set QUANTITY=2


REM --------------------------------------------------
REM STEP 2: ADD PRODUCT TO CART ✅
REM --------------------------------------------------

curl -X POST %CART_BASE_URL%/api/cart/ ^
-H "Authorization: Bearer %CUSTOMER_TOKEN%" ^
-H "Content-Type: application/json" ^
-d "{\"product_id\":%PRODUCT_ID%,\"variant_id\":%VARIANT_ID%,\"name\":\"%PRODUCT_NAME%\",\"color\":\"%COLOR%\",\"price\":%PRICE%,\"quantity\":%QUANTITY%}"


REM --------------------------------------------------
REM STEP 3: VIEW CART PREVIEW ✅
REM --------------------------------------------------

curl -X GET %CART_BASE_URL%/api/cart/ ^
-H "Authorization: Bearer %CUSTOMER_TOKEN%"


REM --------------------------------------------------
REM STEP 4: CHECKOUT
REM --------------------------------------------------

curl -X POST %CART_BASE_URL%/api/checkout/ ^
-H "Authorization: Bearer %CUSTOMER_TOKEN%" ^
-H "Content-Type: application/json" ^
-d "{\"contact\":\"9394113857\",\"address\":\"Hyderabad\"}"


REM --------------------------------------------------
REM STEP 5: VIEW ORDERS
REM --------------------------------------------------

curl -X GET %CART_BASE_URL%/api/orders/ ^
-H "Authorization: Bearer %CUSTOMER_TOKEN%"
